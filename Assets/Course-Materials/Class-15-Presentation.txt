SLIDE 7: Chaining Animations
-----------------------------
```csharp
// Sequential
transform.DOMoveY(5f, 1f)
    .OnComplete(() => transform.DOMoveX(5f, 1f));

// Using Sequence
Sequence mySequence = DOTween.Sequence();
mySequence.Append(transform.DOMoveY(5f, 1f));
mySequence.Append(transform.DOMoveX(5f, 1f));
mySequence.Append(transform.DOScale(2f, 0.5f));
mySequence.Play();
```

SLIDE 11: Squash and Stretch
-----------------------------
- Animation principle
- Makes motion feel alive
- Exaggerate deformation
- Maintain volume

```csharp
// Squash (wider, shorter)
transform.DOScale(new Vector3(1.4f, 0.7f, 1f), 0.1f);

// Stretch (narrower, taller)
transform.DOScale(new Vector3(0.8f, 1.3f, 1f), 0.1f);

// Return to normal
transform.DOScale(Vector3.one, 0.2f);
```

SLIDE 12: Brick Animation Implementation
-----------------------------------------
```csharp
IEnumerator AnimateBrickToPosition(Brick3D brick, Vector3 targetPos, float delay)
{
    yield return new WaitForSeconds(delay);
    
    Vector3 startPos = Camera.main.transform.position;
    brick.transform.position = startPos;
    
    // Animate to target with arc
    float duration = 0.8f;
    float height = 4f;
    
    // Parabolic motion
    for (float t = 0; t < duration; t += Time.deltaTime)
    {
        float progress = t / duration;
        float parabola = height * 4 * progress * (1 - progress);
        
        Vector3 currentPos = Vector3.Lerp(startPos, targetPos, progress);
        currentPos.y += parabola;
        brick.transform.position = currentPos;
        
        // Squash and stretch based on velocity
        ApplySquashStretch(brick, progress);
        
        yield return null;
    }
    
    // Landing squash
    brick.transform.DOScale(new Vector3(1.4f, 0.7f, 1f), 0.05f)
        .OnComplete(() => {
            brick.transform.DOScale(Vector3.one, 0.2f).SetEase(Ease.OutBounce);
        });
}
```

SLIDE 14: Pattern Implementation
---------------------------------
```csharp
bool ShouldPlaceBrick(int x, int y, WallPattern pattern)
{
    switch (pattern)
    {
        case WallPattern.Normal:
            return true;
            
        case WallPattern.Checkerboard:
            return (x + y) % 2 == 0;
            
        case WallPattern.Diagonal:
            return (x + y) % 3 != 0;
            
        case WallPattern.Pyramid:
            return Mathf.Abs(x - columns/2) <= y;
            
        case WallPattern.Circle:
            Vector2 center = new Vector2(columns/2, rows/2);
            float distance = Vector2.Distance(new Vector2(x, y), center);
            return distance <= rows/2;
            
        default:
            return true;
    }
}
```

SLIDE 18: Optimization Techniques
----------------------------------
1. Object Pooling:
   - Reuse objects instead of Instantiate/Destroy
   - Reduces garbage collection

2. Culling:
   - Don't render off-screen objects
   - Disable distant physics

3. LOD (Level of Detail):
   - Simpler models at distance
   - Fewer particles far away

4. Batching:
   - Combine similar objects
   - Static batching for static objects

SLIDE 19: Object Pooling Example
---------------------------------
```csharp
public class ObjectPool : MonoBehaviour
{
    public GameObject prefab;
    private List<GameObject> pool = new List<GameObject>();
    
    public GameObject Get()
    {
        foreach (GameObject obj in pool)
        {
            if (!obj.activeInHierarchy)
            {
                obj.SetActive(true);
                return obj;
            }
        }
        
        // Create new if none available
        GameObject newObj = Instantiate(prefab);
        pool.Add(newObj);
        return newObj;
    }
    
    public void Return(GameObject obj)
    {
        obj.SetActive(false);
    }
}
```
