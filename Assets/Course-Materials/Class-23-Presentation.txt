SLIDE 4: Pattern Implementation
--------------------------------
```csharp
bool ShouldPlaceBrick(int col, int row)
{
    switch (currentPattern)
    {
        case WallPattern.Normal:
            return true;
            
        case WallPattern.Checkerboard:
            return (col + row) % 2 == 0;
            
        case WallPattern.Diagonal:
            return (col + row) % 3 != 0;
            
        case WallPattern.Pyramid:
            int center = columns / 2;
            int halfWidth = row + 1;
            return col >= center - halfWidth && col <= center + halfWidth;
            
        default:
            return true;
    }
}
```

SLIDE 5: More Patterns
-----------------------
```csharp
case WallPattern.Circle:
    float centerX = columns / 2f;
    float centerY = rows / 2f;
    float distance = Vector2.Distance(
        new Vector2(col, row), 
        new Vector2(centerX, centerY)
    );
    return distance <= Mathf.Min(rows, columns) / 2f;

case WallPattern.Random:
    return Random.value > 0.2f;  // 80% chance
```

SLIDE 7: Weather Effects
-------------------------
Add atmosphere:
```csharp
[Header("Weather")]
[SerializeField] private GameObject rainEffect;
[SerializeField] private GameObject snowEffect;

void ActivateRandomWeather()
{
    // Disable all
    rainEffect.SetActive(false);
    snowEffect.SetActive(false);
    
    // Random selection
    int weather = Random.Range(0, 3);
    
    if (weather == 0) rainEffect.SetActive(true);
    else if (weather == 1) snowEffect.SetActive(true);
    // else no weather
}
```

SLIDE 8: Progressive Difficulty
--------------------------------
```csharp
private int wallIndex = 0;

float GetBrickMass()
{
    float minMass = 30f;
    float maxMass = 400f;
    float progress = (float)wallIndex / 10f;  // 10 walls to max
    progress = Mathf.Clamp01(progress);
    
    return Mathf.Lerp(minMass, maxMass, progress);
}
```

SLIDE 9: Object Pooling Concept
--------------------------------
Problem:
- Instantiate/Destroy is expensive
- Creates garbage collection
- Can cause frame drops

Solution:
- Pre-create objects
- Reuse instead of destroy
- "Pool" of available objects

SLIDE 10: Simple Object Pool
-----------------------------
```csharp
public class ObjectPool : MonoBehaviour
{
    [SerializeField] private GameObject prefab;
    [SerializeField] private int initialSize = 20;
    
    private Queue<GameObject> pool = new Queue<GameObject>();
    
    void Start()
    {
        for (int i = 0; i < initialSize; i++)
        {
            GameObject obj = Instantiate(prefab);
            obj.SetActive(false);
            pool.Enqueue(obj);
        }
    }
    
    public GameObject Get()
    {
        if (pool.Count > 0)
        {
            GameObject obj = pool.Dequeue();
            obj.SetActive(true);
            return obj;
        }
        return Instantiate(prefab);
    }
    
    public void Return(GameObject obj)
    {
        obj.SetActive(false);
        pool.Enqueue(obj);
    }
}
```

SLIDE 11: Using Object Pool
----------------------------
```csharp
// Instead of Instantiate
```

- Large textures

SLIDE 15: Optimization Tips
----------------------------
```csharp
// Cache references
private Transform cachedTransform;
void Awake() { cachedTransform = transform; }

// Avoid Find in Update
void Start() { player = GameObject.Find("Player"); }

// Use CompareTag
if (other.CompareTag("Enemy"))  // Good
if (other.tag == "Enemy")       // Bad

// Avoid string concatenation in Update
Debug.Log($"Score: {score}");  // Creates garbage
```
